# Покажи пример запроса создания таблицы users и orders?

Покажи пример запроса создания таблицы users и orders?::


-- Создание таблицы a_users
```

CREATE TABLE a_users (
    -- Столбец user_id: целочисленный идентификатор пользователя, который автоматически увеличивается при добавлении новых строк.
    -- Он также является первичным ключом таблицы, что гарантирует уникальность и позволяет эффективно индексировать строки.
    user_id INT AUTO_INCREMENT PRIMARY KEY,

    -- Столбец username: строковое значение для имени пользователя, длиной до 50 символов.
    -- Этот столбец не может содержать NULL значения, что обеспечивается ограничением NOT NULL.
    username VARCHAR(50) NOT NULL,

    -- Столбец email: строковое значение для электронной почты пользователя, длиной до 100 символов.
    -- Этот столбец также не может содержать NULL значения и имеет уникальное ограничение UNIQUE,
    -- что означает, что все значения в этом столбце должны быть уникальными.
    email VARCHAR(100) NOT NULL UNIQUE,

    -- Столбец created_at: временная метка, которая автоматически получает текущее время при создании записи.
    -- Значение по умолчанию устанавливается на текущее время при добавлении новой строки в таблицу.
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

-- Создание таблицы orders
```

CREATE TABLE orders (
    -- Столбец order_id: целочисленный идентификатор заказа, который автоматически увеличивается при добавлении новых строк.
    -- Он также является первичным ключом таблицы, что гарантирует уникальность и позволяет эффективно индексировать строки.
    order_id INT AUTO_INCREMENT PRIMARY KEY,

    -- Столбец user_id: целочисленный идентификатор пользователя, который связан с таблицей a_users.
    -- Это поле будет использоваться для связи заказа с конкретным пользователем.
    user_id INT,

    -- Столбец order_date: дата заказа.
    -- Этот столбец хранит дату в формате DATE, без времени.
    order_date DATE,

    -- Столбец total_amount: общая сумма заказа.
    -- Этот столбец хранит числовое значение с двумя знаками после запятой.
    total_amount DECIMAL(10, 2),

    -- Создание внешнего ключа (FOREIGN KEY) для связи с таблицей a_users.
    -- Это ограничение гарантирует, что значение в столбце user_id таблицы orders
    -- должно соответствовать существующему значению в столбце user_id таблицы a_users.
    FOREIGN KEY (user_id) REFERENCES a_users(user_id)
);
```
