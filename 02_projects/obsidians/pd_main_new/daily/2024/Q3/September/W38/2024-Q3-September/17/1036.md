<?php  
  
  
//use App\Development\TypeScript\FilterAngular;  
//use App\Development\TypeScript\FilterConfigurator;  
  
use App\Analytics\ManagersBlocks\Filter;  
use App\Analytics\ManagersBlocks\FilterConfigurator;  
use App\Development\TypeScript\FilterConfiguratorInterface;  
use App\Filter\Controller\AbstractDataSourceController;  
  
/**  
 * Class Development_TypeScript_FilterAngularController */class Analitics_ManagersBlocks_ListController extends AbstractDataSourceController  
{  
  
    /**  
     * @inheritDoc     * @return Filter  
     */  
    public function createFilter()  
    {  
        return (new Filter())->setShowSql(true)->debugSql();  
    }  
  
  
    /**  
     * @inheritDoc     * @return FilterConfiguratorInterface|null  
     */  
    public function getFilterConfigurator(): ?FilterConfiguratorInterface  
    {  
        return new FilterConfigurator();  
    }  
  
  
    /**  
     * @inheritDoc     * @return array  
     */  
    protected function _getFilterModel()  
    {  
        if (($id = (int)$this->_getParam('searchId'))) {  
            return [  
                'searchId' => [  
                    'type'   => 'equals',  
                    'filter' => $id,  
                ],  
            ];  
        }  
  
        $filter   = $this->getFilter();  
        $maskData = $this->_getParam('filterModel', []);  
  
  
        $filter->setExpired((bool)$this->_getParam('expired'));  
        $filter->setFinanceDepot((bool)$this->_getParam('finance_depot'));  
  
        return $maskData;  
    }  
  
}
