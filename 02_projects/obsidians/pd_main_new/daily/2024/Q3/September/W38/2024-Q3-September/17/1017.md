```PHP
<?php  
  
  
use App\Controller\AngularJS\AbstractFilterPageController;  
  
use App\Analytics\ManagersBlocks\Filter;  
use App\Analytics\ManagersBlocks\FilterConfigurator;  
use App\Development\TypeScript\FilterConfiguratorInterface;  
  
  
class Analitics_ManagersBlocks_ListController extends AbstractFilterPageController  
{  
  
    protected $filterConfigurator;  
    public function createFilter()  
    {  
        $filter = new Filter();  
//        $this->_getFilterConfigurator()->setUpFilter($filter);  
//        return $filter;  
        return new Filter();  
    }  
  
    public function getFilterConfigurator(): ?\App\Development\TypeScript\FilterConfiguratorInterface  
    {  
        if(!($this->filterConfigurator instanceof FilterConfigurator)) {  
            $this->filterConfigurator = new FilterConfigurator();  
        }  
  
        return $this->filterConfigurator;  
  
    }  
  
    public function indexAction()  
    {  
        $this->view->getRequire()  
            ->setDataMain('/js/app/application/analytics/managers-block/list/index/config')  
            ->setParams('page.module', [  
                'dataTable' => [  
                    'urls' => $this->_getDatatableRoutes(),  
                ],  
                'access' => [  
                    'managersblocks' => [  
                        'refer' => App_Access::get('all', 'analitics>managersblocks>refer'),  
                    ]  
                ],  
            ]);  
    }  
  
    public function init()  
    {  
        parent::init();  
        Zend_Registry::set('statusString', 'Брони заказов');  
        \App\Access\Guard::assertEqual('analitics>managersblocks>refer', true);  
    }  
  
  
  
}
```
