namespace App\Analytics\ProductionDateDoneShift;  
  
use App\Development\TypeScript\Mask\DateMask;  
use App\Development\TypeScript\Mask\MaskInterface;  
use App\Development\TypeScript\Mask\NumberMask;  
use App\Development\TypeScript\Mask\Options\DbOptions;  
use App\Development\TypeScript\Mask\StringMask;  
use App\Development\TypeScript\WorkPeriod\WorkPeriodModel2;  
  
class Filter extends \App\Development\TypeScript\AbstractFilter  
{  
  
  
    /**  
     * @var \App\Development\TypeScript\WorkPeriod\WorkPeriodModel2  
     */  
    private $workPeriod;  
  
    /**  
     * @return \App\Development\TypeScript\WorkPeriod\WorkPeriodModel2  
     */  
    public function getWorkPeriod(): \App\Development\TypeScript\WorkPeriod\WorkPeriodModel2  
    {  
        return $this->workPeriod;  
    }  
  
    public function init($params = null): \App\Development\TypeScript\AbstractFilter  
    {  
        $this->setFilterResult(new FilterResult(['filter' => $this]));  
  
        $before = (new \DateTimeImmutable())->setTime(0, 0, 0, 0);  
        $this->workPeriod = new WorkPeriodModel2(static::class, [  
            'dateBefore' => $before,  
            'dateAfter'  => $before->modify('-3 month'),  
        ]);  
        $this->workPeriod->init();  
  
        $whereDecorator = new \App_Filter_Component_Decorator_Type_DefaultWhereConditions_Decorator();  
        $whereDecorator->getWhereConditions()->add("claims.type_id = " . \App_Claim_Factory::TYPE_ID_PRODUCTION);  
        $whereDecorator->getWhereConditions()->add($this->workPeriod->getCondition('claims', 'date_begin'));  
        $this->addOldDecorator($whereDecorator);  
  
        $this->addClaimIdMask();  
        $this->addClientMask();  
        $this->addManufacturerMask();  
        $this->addFirsDateDoneMask();  
        $this->addCurrentDateDoneMask();  
        $this->addDaysInWorkMask();  
        $this->addDateDoneShift();  
        $this->addDateReady();  
        $this->addProductFieldsMasks();  
  
        return parent::init($params);  
    }  
  
    /**  
     * @inheritDoc     */    public function getResultSql(): string  
    {  
        $configQuery = \App_Config_Depot::getQueryPartProduction([1,2,3,4,5,6,7,8,15]);  
  
        return "  
            SELECT                claims.`full_id`,  
                claims.url AS claim_url,                claims.id AS claim_id,      
                  
                IF(  
                    clients.id,                    clients.`s_title`,                    claim_production_textclientname.`client_name`                ) AS client_title,                                claims.claim_status,  
                claims.date_closed,                                claim_production_manufacturer.`manufacturer_id`,  
                manufacturers.s_title AS manufacturer,                             claim_production_manufacturer_date_done.date_done AS first_date_done,  
                claim_production_manufacturer.date_done AS current_date_done,                                DATEDIFF(  
                    IF(                        claims.claim_status = 0,                        claims.date_set_in_work_pending_status,   
                        CURDATE()  
                    ),                    FROM_UNIXTIME(claims.`date_begin`)  
                ) AS days_in_work,                                DATEDIFF(  
                    claim_production_manufacturer.date_done,                    claim_production_manufacturer_date_done.date_done                ) AS date_done_shift,                                JSON_ARRAYAGG(  
                    JSON_OBJECT(                        'url', COALESCE(claim_in.url, ''),                        'date', FROM_UNIXTIME(claim_in.date, '%Y-%m-%d'),                        'full_id', COALESCE(claim_in.full_id, '')                    )                ) AS claims_in,  
                IFNULL(cpp.product_id, 0) AS product_id,                {$configQuery}  
            FROM claims  
            INNER JOIN claim_production_product AS cpp ON                cpp.claim_id = claims.id  
            LEFT JOIN clients ON                claims.client_id = clients.id  
            LEFT JOIN `claim_production_textclientname` ON                claims.id = claim_production_textclientname.`claim_id`  
            INNER JOIN claim_production_manufacturer ON                claims.id = claim_production_manufacturer.claim_id  
            INNER JOIN clients_manufacturers ONclaim_production_manufacturer.`manufacturer_id` = clients_manufacturers.`id`  
            INNER JOIN clients AS manufacturers ON                `clients_manufacturers`.`client_id` = manufacturers.`id`  
            LEFT JOIN claim_production_manufacturer_result AS cpmr ON                cpmr.cpm_id = claim_production_manufacturer.id  
            LEFT JOIN claim_production_manufacturer_result_products AS cpmrp ON                cpmrp.cpmr_id = cpmr.id  
            LEFT JOIN claim_products AS claim_in_cp ON                claim_in_cp.id = cpmrp.cp_id  
            LEFT JOIN claims AS claim_in ON                claim_in.id = claim_in_cp.claim_id  
            LEFT JOIN (                SELECT                    cpm_id,  
                    date_done  
                FROM claim_production_manufacturer_date_done  
                WHERE id = (                    SELECT                        MIN(t1.id) AS min_id  
                    FROM claim_production_manufacturer_date_done AS t1  
                    WHERE claim_production_manufacturer_date_done.cpm_id = t1.`cpm_id`  
                )            ) AS claim_production_manufacturer_date_done ON                claim_production_manufacturer.id = claim_production_manufacturer_date_done.`cpm_id`  
            {WHERE}            GROUP BY claims.id, claim_production_manufacturer.`manufacturer_id`        ";  
    }  
  
    private function addClaimIdMask()  
    {  
        $this->addMask(  
            (new NumberMask())  
                ->setCaption('Заявка')  
                ->setName('claim_id')  
                ->setTarget("claims.id")  
                ->setOptions("  
                    SELECT DISTINCT                        claims.id AS v,  
                        claims.`full_id` AS t                    FROM claims                    INNER JOIN claim_production_manufacturer ON                        `claims`.`id` = `claim_production_manufacturer`.`claim_id`  
                    ORDER BY t                ")  
                ->setFilterUse(MaskInterface::FILTER_USE_SELECT)  
        );  
    }  
  
    private function addClientMask()  
    {  
        $this->addMask(  
            (new StringMask())  
                ->setCaption('Клиент')  
                ->setName('client_title')  
                ->setTarget(['HAVING' => 'client_title'])  
                ->setOptions("  
                    SELECT *                    FROM (                        SELECT DISTINCT                            `c`.`s_title` AS v,                            `c`.`s_title` AS t                        FROM                            `claims` AS `cl`                        JOIN `clients` AS `c` ON `c`.`id` = `cl`.`client_id`                        WHERE`cl`.`type_id` = " . \App_Claim_Factory::TYPE_ID_PRODUCTION . "  
                        UNION                        SELECT DISTINCT                            `client_name` AS v,                            `client_name` AS t                        FROM `claim_production_textclientname`                    ) AS t                    ORDER BY t                ")  
                ->setFilterUse(MaskInterface::FILTER_USE_SELECT)  
        );  
    }  
  
    private function addManufacturerMask()  
    {  
        $this->addMask(  
            (new NumberMask())  
                ->setCaption('Площадка')  
                ->setName('manufacturer_id')  
                ->setTarget('claim_production_manufacturer.manufacturer_id')  
                ->setOptions("  
                    SELECT DISTINCT                        clients_manufacturers.`id` AS v,  
                        clients.s_title AS t                    FROM `clients_manufacturers`                    INNER JOIN `clients` ON                        `clients_manufacturers`.`client_id` = `clients`.`id`  
                    ORDER BY t                ")  
                ->setFilterUse(MaskInterface::FILTER_USE_SELECT)  
        );  
    }  
  
    private function addFirsDateDoneMask()  
    {  
        $this->addMask(  
            (new DateMask())  
                ->setCaption('Первоначальный срок')  
                ->setName('first_date_done')  
                ->setMaskUse(false)  
                ->setTarget('claim_production_manufacturer_date_done.date_done')  
        );  
    }  
  
    private function addCurrentDateDoneMask()  
    {  
        $this->addMask(  
            (new DateMask())  
                ->setCaption('Текущий срок')  
                ->setName('current_date_done')  
                ->setMaskUse(false)  
                ->setTarget('claim_production_manufacturer.date_done')  
        );  
    }  
  
    private function addDaysInWorkMask()  
    {  
        $this->addMask(  
            (new NumberMask())  
                ->setCaption('В работе')  
                ->setName('days_in_work')  
                ->setTarget(['HAVING' => 'days_in_work'])  
                ->setOptions((new DbOptions("  
                    SELECT DISTINCT                        DATEDIFF(  
                            IF(                                claims.claim_status = 0,                                claims.date_set_in_work_pending_status,  
                                CURDATE()                            ),                            FROM_UNIXTIME(claims.`date_begin`)  
                        ) AS v,                                                DATEDIFF(  
                            IF(                                claims.claim_status = 0,                                claims.date_set_in_work_pending_status,  
                                CURDATE()                            ),                            FROM_UNIXTIME(claims.`date_begin`)  
                        ) AS t                    FROM claims                    INNER JOIN claim_production_product AS cpp ON                        cpp.claim_id = claims.id                    INNER JOIN claim_production_manufacturer ON                        claims.id = claim_production_manufacturer.claim_id                    INNER JOIN clients_manufacturers ON                        claim_production_manufacturer.`manufacturer_id` = clients_manufacturers.`id`  
                    LEFT JOIN (                        SELECT                            cpm_id,                            date_done                        FROM claim_production_manufacturer_date_done                        WHERE id = (                            SELECT                                MIN(t1.id) AS min_id                            FROM claim_production_manufacturer_date_done AS t1                            WHERE claim_production_manufacturer_date_done.cpm_id = t1.`cpm_id`                        )                    ) AS claim_production_manufacturer_date_done ON                        claim_production_manufacturer.id = claim_production_manufacturer_date_done.`cpm_id`                    {WHERE}                    ORDER BY t                "))->addDecorator($this->getOptionsQueryDecorator()))  
                ->setFilterUse(MaskInterface::FILTER_USE_COMBO)  
        );  
    }  
  
    private function addDateDoneShift()  
    {  
        $this->addMask(  
            (new NumberMask())  
                ->setCaption('Кол-во дней переноса')  
                ->setName('date_done_shift')  
                ->setTarget(['HAVING' => 'date_done_shift'])  
                ->setOptions((new DbOptions("  
                    SELECT DISTINCT                        DATEDIFF(  
                            claim_production_manufacturer.date_done,                            claim_production_manufacturer_date_done.date_done                        ) AS v,                        DATEDIFF(                            claim_production_manufacturer.date_done,                            claim_production_manufacturer_date_done.date_done                        ) AS t                    FROM claims                    INNER JOIN claim_production_product AS cpp ON                        cpp.claim_id = claims.id                    INNER JOIN claim_production_manufacturer ON                        claims.id = claim_production_manufacturer.claim_id                    INNER JOIN clients_manufacturers ON                        claim_production_manufacturer.`manufacturer_id` = clients_manufacturers.`id`  
                    LEFT JOIN (                        SELECT                            cpm_id,                            date_done                        FROM claim_production_manufacturer_date_done                        WHERE id = (                            SELECT                                MIN(t1.id) AS min_id                            FROM claim_production_manufacturer_date_done AS t1                            WHERE claim_production_manufacturer_date_done.cpm_id = t1.`cpm_id`                        )                    ) AS claim_production_manufacturer_date_done ON                        claim_production_manufacturer.id = claim_production_manufacturer_date_done.`cpm_id`                    {WHERE}                    ORDER BY t                "))->addDecorator($this->getOptionsQueryDecorator()))  
                ->setFilterUse(MaskInterface::FILTER_USE_COMBO)  
        );  
    }  
  
    private function addDateReady(): void  
    {  
        $this->addMask(  
            new DateMask()  
                ->setCaption('Фактическая дата готовности')  
                ->setName('date_ready')  
                ->setTarget("FROM_UNIXTIME(claim_in.date, '%Y-%m-%d')")  
        );  
    }  
  
    private function addProductFieldsMasks()  
    {  
        $this->addMask(  
            (new NumberMask())  
                ->setNullable(true)  
                ->setNullableValue(0)  
                ->setCaption('id товара')  
                ->setName('product_id')  
                ->setTarget("`cpp`.`product_id`")  
                ->setOptions("  
                    SELECT 0 AS v, 'пусто' AS t                    UNION ALL                    SELECT *                    FROM (                        SELECT DISTINCT                            `cpp`.`product_id` AS `v`,                            `cpp`.`product_id` AS `t`                        FROM `claim_production_product` AS `cpp`                        ORDER BY `t`                    ) AS t                ")  
                ->setFilterUse(MaskInterface::FILTER_USE_SELECT)  
        );  
  
        // поля склада  
        for ($i = 1; $i < 9; $i++) {  
            $field = \App_Config::get('depot')->{'field' . $i};  
  
            $mask = (new NumberMask())  
                ->setCaption($field->title)  
                ->setName('field' . $i)  
                ->setTarget('cpp.field' . $i)  
                ->setFilterUse(MaskInterface::FILTER_USE_SELECT)  
            ;  
  
            if ($field->filtertype === 'string') {  
                $mask->setOptions("  
                    SELECT                        `id` AS v,  
                        `text` AS t                    FROM `claim_production_product_fields`                    WHERE `field_id` = {$i}  
                    ORDER BY t  
                ");  
            } else if ($field->filtertype === 'number') {  
                $mask->setOptions("  
                    SELECT DISTINCT                        field{$i} AS v,  
                        field{$i} AS t  
                    FROM `claim_production_product`                    ORDER BY t  
                ");  
            }  
  
            $this->addMask($mask);  
        }  
    }  
}
