<?php  
  
use Amd\AmdCoreServices\Container\ContainerService;  
use App\Claim\Production\ProcessRequest\Service\CrudService;  
use App\Claim\Production\RepeatedAppeals\Models\RepeatedAppealModel;  
use App\Claim\Production\RepeatedAppeals\Models\RepeatedAppealReasonModel;  
use App\Claim\Production\RepeatedAppeals\Repositories\Db\RepeatedAppealModel\RepeatedAppealConditions;  
use App\Claim\Production\RepeatedAppeals\Repositories\Db\RepeatedAppealReasonModel\RepeatedAppealReasonConditions;  
use App\Claim\Production\RepeatedAppeals\Repositories\RepeatedAppealReasonsRepository;  
use App\Claim\Production\RepeatedAppeals\Repositories\RepeatedAppealsRepository;  
  
/**  
 * Mapper для модели заявки типа "Заявка на производство" * * @category    App * @package     Claim * @subpackage  Mapper * * @author Anisimov A. <sanchezgeek@gmail.com>  
 */class App_Claim_Mapper_Production  
{  
    /**  
     * Класс таблицы claims     *     * @var Claim_Model_DbTable_Depot2_ClaimTable  
     */  
    protected $_claimTable;  
  
    /**  
     * Репозиторий сущностей повторных обращений     * @var RepeatedAppealsRepository  
     */  
    private $repeatedAppealsRepository;  
  
    /**  
     * Репозиторий сущностей причин повторных обращений     * @var RepeatedAppealReasonsRepository  
     */  
    private $repeatedAppealReasonsRepository;  
  
    /**  
     * CRUD сервис для обращений на обработку     * @var CrudService  
     */  
    private $processRequestCrudService;  
  
    /**  
     * @throws Zend_Exception  
     * @throws \DI\DependencyException  
     * @throws \DI\NotFoundException  
     */  
    public function __construct()  
    {  
        /** @var ContainerService $container */  
        $container                             = Zend_Registry::get('container');  
        $this->repeatedAppealsRepository       = $container->get(RepeatedAppealsRepository::class);  
        $this->repeatedAppealReasonsRepository = $container->get(RepeatedAppealReasonsRepository::class);  
        $this->processRequestCrudService       = $container->get(CrudService::class);  
    }  
  
    /**  
     * Возвращает класс таблицы claims     *     * @return App_Db_Claims  
     */  
    public function getClaimTable()  
    {  
        if (null === $this->_claimTable) {  
            $this->_claimTable = App_Db::get(DB_CLAIMS);  
        }  
        return $this->_claimTable;  
    }  
  
  
    /**  
     * Сохранение заявки     *     * @param Claim_Model_Production $claim Объект заявки  
     * @throws Zend_Db_Adapter_Exception  
     * @throws Zend_Db_Statement_Exception  
     * @throws Zend_Exception  
     */  
    public function save(Claim_Model_Production &$claim)  
    {  
        // Основные данные по заявке  
        $data = array();  
  
        if (Claim_Model_Production_Access::getFieldPermission($claim->id, 'depot_index') > App_Access_Field_Type2::ACCESS_DENIED ||  
            $claim->claimIsCopying  
        ) {  
            $data['depot_index'] = (int) $claim->depotIndex;  
        } else if (!$claim->id) {  
            $data['depot_index'] = 2;  
        }  
  
        if (Claim_Model_Production_Access::canEditField($claim->id, 'client') || $claim->claimIsCopying) {  
            $data['client_id'] = is_null($claim->clientId) ? 0 : $claim->clientId;  
        }  
  
        if (!$claim->id && $claim->claimIsCopying) {  
            // В случае сохранения копирования, менеджер - текущий пользователь  
            $data['manager_id'] = (integer) Zend_Auth::getInstance()->getIdentity()->id;  
        } else if (Claim_Model_Production_Access::canEditField($claim->id, 'manager')) {  
            $data['manager_id'] = $claim->managerId;  
        } else if (!(integer) $claim->id) {  
            $data['manager_id'] = (integer) Zend_Auth::getInstance()->getIdentity()->id;  
        }  
  
        if ($claim->claimStatus == App_Claim_Status_Production::CREATE_STATUS && $claim->changeStatusAvaleble) {  
            // При отправке заявки со статуса создания, необходимо указать сегодняшнюю дату  
            $data['date'] = App_Formatter::date2int(date('d.m.Y'));  
  
            $resetDate = true;  
        } else if (  
            $claim->changeStatusAvaleble  
            && $claim->claimStatus == App_Claim_Status_Production::BASE_PRICE_PROCESSED_STATUS  
            && !$claim->closeReason  
        ) {  
            // Также, при отправке заявки дальше со статуа "Запрос БЦ (у менеджера)"  
            $data['date'] = App_Formatter::date2int(date('d.m.Y'));  
  
            $resetDate = true;  
        } else if (Claim_Model_Production_Access::canEditField($claim->id, 'date') || $claim->claimIsCopying) {  
            $data['date'] = App_Formatter::date2int($claim->date);  
        }  
  
        if (Claim_Model_Production_Access::canEditField($claim->id, 'dateready') || $claim->claimIsCopying) {  
            $data['date_ready'] = $claim->date_ready ? App_Formatter::date2int($claim->date_ready) : 0;  
        }  
  
        if (App_Access_Field_Type2::ACCESS_EDIT === Claim_Model_Production_Access::getFieldPermission($claim->id, 'baseprice_deadline_by_client')) {  
            $data['baseprice_deadline_by_client'] = $claim->getBasepriceDeadlineByClient();  
        }  
  
        if (Claim_Model_Production_Access::canEditField($claim->id, 'manufacturerdateready') || $claim->claimIsCopying) {  
            $data['manufacturer_date_ready'] = $claim->manufacturer_date_ready ? App_Formatter::date2int($claim->manufacturer_date_ready) : 0;  
        }  
  
        /* --------- Блокировать товар под менеджера / на склад ------------- */  
        if (!$claim->id) {  
            // В случае добавление новой заявки, по умолчанию выставляем 1  
            $data['block_for_manager'] = 1;  
        }  
        if (Claim_Model_Production_Access::canEditField($claim->id, 'blockformanager') || $claim->claimIsCopying) {  
            $data['block_for_manager'] = (string) $claim->blockForManager;  
            $data['incoming_production_notification'] = $claim->blockForManager ? 0 : $claim->incomingProductionNotification;  
        }  
        /* ------------------------------------------------------------------ */  
  
        // Конкретно это поле можно сохранять при ACCESS_VIEW_AS_HIDDEN        if (  
            Claim_Model_Production_Access::getFieldPermission($claim->id, 'annotation') >= App_Access_Field_Type3::ACCESS_VIEW_AS_HIDDEN  
            || $claim->claimIsCopying  
        ) {  
            $data['annotation'] = strip_tags($claim->annotation);  
        }  
  
        if (Claim_Model_Production_Access::canEditField($claim->id, 'stmanagerannotation') || $claim->claimIsCopying) {  
            $data['stmanager_annotation'] = strip_tags($claim->stmanager_annotation);  
        }  
  
        if (Claim_Model_Production_Access::canEditField($claim->id, 'purchasingannotation') || $claim->claimIsCopying) {  
            $data['purchasing_annotation'] = strip_tags($claim->purchasing_annotation, '<p><b><strong><em><i><ul><ol>');  
        }  
  
        if (Claim_Model_Production_Access::canEditField($claim->id, 'internalpurchasingannotation') || $claim->claimIsCopying) {  
            $data['internal_purchasing_annotation'] = strip_tags($claim->getInternalPurchasingAnnotation());  
            $data['internal_purchasing_annotation_last_edition'] = $claim->getInternalPurchasingAnnotationLastEdition();  
  
        }  
  
        if (Claim_Model_Production_Access::canEditField($claim->id, 'technologist_annotation') || $claim->getClaimIsCopyingFlag()) {  
            $data['technologist_annotation'] = strip_tags($claim->getTechnologistAnnotation());  
            $data['technologist_annotation_last_edition'] = $claim->getTechnologistAnnotationLastEdition();  
  
        }  
  
        if ($claim->changeStatusAvaleble && (int) $claim->claimStatus !== App_Claim_Status_Production::STMANAGER_STATUS_AFTER_BASE_PRICE_IN_PROCESS_STATUS) {  
            // Если заявка отправляется, то надо сбросить флаг closeReason. Если он есть в POST, то позже запишется новый.  
            // Игнор в случае статуса "подтверждение рук. отдела закупок (ответ по запросу БЦ)".            $data['close_reason'] = new Zend_Db_Expr('NULL');  
        }  
  
        if (  
            $claim->claimStatus == App_Claim_Status_Production::CREATE_STATUS // если заявка на статусе создания  
            && $claim->changeStatusAvaleble // и заявка отправляется  
        ) {  
            // то надо проверить указан ли статус, на который отправляет заявку менеджер  
            if (!(  
                strlen($claim->userSendToStatusId)  
                && (  
                    in_array(intval($claim->userSendToStatusId), array(  
                        App_Claim_Status_Production::BASE_PRICE_IN_PROCESS_STATUS,  
                        App_Claim_Status_Production::IN_WORK_PENDING_STATUS  
                    )) || (  
                        // В случае если заявка уже сохранена, есть возможность закрыть заявку  
                        $claim->id > 0  
                        && intval($claim->userSendToStatusId) === App_Claim_Status_Production::CLOSE_STATUS  
                    )  
                )  
            )) {  
                throw new Exception('Неверно указан статус для дальнейшего движения заявки');  
            }  
  
            // Флаг отправки заявки менеджером на определённый статус  
            $data['manager_send_to_status_id'] = $claim->userSendToStatusId;  
        }  
  
        if (  
            $claim->claimStatus == App_Claim_Status_Production::BASE_PRICE_IN_PROCESS_STATUS // если заявка на статусе "запрос БЦ (в обработке)"  
            && $claim->changeStatusAvaleble // и заявка отправляется  
        ) {  
            // то надо проверить указан ли статус, на который пользователь отправляет заявку  
            if (!(  
                strlen($claim->userSendToStatusId)  
                && in_array($claim->userSendToStatusId, array(  
                    App_Claim_Status_Production::BASE_PRICE_PROCESSED_STATUS,  
                    App_Claim_Status_Production::CLOSE_STATUS,  
                    App_Claim_Status_Production::CREATE_STATUS  
                ))  
            )) {  
                throw new Exception('Неверно указан статус для дальнейшего движения заявки');  
            }  
  
            // Статус, на который отправлена заявка после ответа по запросу БЦ  
            $data['talib_send_to_status_id'] = $claim->userSendToStatusId;  
        }  
  
        if ($claim->closeReason) {  
            // Если указана причина закрытия заявки, запишем её в массив данных для сохранения/обновления  
            if (Claim_Model_Production_CloseReasons::checkCloseReason($claim->closeReason)) {  
                $data['close_reason'] = $claim->closeReason;  
            }  
        }  
  
        $statusClass = new App_Claim_Status_Production($claim->id, $claim);  
        if ($claim->changeStatusAvaleble && $statusClass->getNextStatus() == App_Claim_Status_Production::IN_WORK_PENDING_STATUS) {  
            // Если заявка отправляется на статус "В работе (ожидание)", запишем дату поступления заявки в работу  
            $now_date = date_create();  
            $data['date_come_in_work'] = $now_date->format('Y-m-d H:i:s');  
        } else if ($claim->changeStatusAvaleble && $statusClass->getNextStatus() == App_Claim_Status_Production::BASE_PRICE_IN_PROCESS_STATUS) {  
            // Если заявка отправляется на статус "Запрос БЦ (в обработке)", запишем дату/время запроса БЦ  
            $data['date_baseprice_requested'] = time();  
        }  
  
        if ($claim->copiedFromClaimId) {  
            // Исходная заявка  
            $data['copied_from_claim_id'] = $claim->copiedFromClaimId;  
        }  
  
        if ($claim->productValuesIsChanged) {  
            // При сбросе БЦ сбрасываем также и поле "Занялся обработкой"  
            $data['baseprice_process_user_id'] = new Zend_Db_Expr('NULL');  
  
            //сбросить поле processor_id у обращения на обработку БЦ, найденного по claimId  
            $this->processRequestCrudService->setProcessorByClaimId($claim->id, null);  
        }  
  
        if ((int) Claim_Model_Production_Access::getFieldPermission($claim->id, 'ignoreclientproductionlimit') === 2) {  
            $data['unblock_send_button'] = $claim->unblockSendButton ? $claim->unblockSendButton : 0;  
        }  
  
        if ((int) Claim_Model_Production_Access::getFieldPermission($claim->id, 'need_gendir_consideration') === App_Access_Field_Type3::ACCESS_EDIT) {  
            $data['need_gendir_consideration'] = (int) $claim->needGendirConsideration;  
        }  
  
        if ((int) Claim_Model_Production_Access::getFieldPermission($claim->id, 'annotation_gendir') === App_Access_Field_Type2::ACCESS_EDIT) {  
            $data['annotation_gendir'] = $claim->getAnnotationGendir();  
        }  
  
        if ((int) Claim_Model_Production_Access::getFieldPermission($claim->id, 'need_technologist_consideration') === App_Access_Field_Type3::ACCESS_EDIT) {  
            $data['need_technologist_consideration'] = (int) $claim->needTechnologistConsideration;  
        }  
  
        // AMD-583, если загружают файл с примечаниями технологу, выставить флаг "need_technologist_consideration"  
        // Заявка на создании + отправляют заявку (меняется статус) + есть данные о файле        if ((int) $claim->claimStatus === App_Claim_Status_Production::CREATE_STATUS &&  
            (int) $claim->changeStatusAvaleble &&  
            (  
                isset($_POST['production_params']['comments_for_technologist']) ||  
                !empty($_FILES['production_params']['name']['comments_for_technologist'][0]['uploaded_file_names'][0])  
            )  
        ) {  
            $data['need_technologist_consideration'] = 1;  
        }  
  
        if ($claim->claimIsCopying) {  
            $data['production_waiting_closure'] = new Zend_Db_Expr('NULL');  
            $data['production_return_annotation'] = '';  
        } else {  
            // если менеджер хочет закрыть заявку, выставляем флаг об ожидании подтверждения закрытия  
            $data['production_waiting_closure'] = $claim->productionWaitingClosure;  
        }  
  
        if (!$claim->id) {  
            // Добавление новой заявки  
            $data['dispetcherFlag'] = 0;  
            $data['talibFlag'] = 0;  
            $data['claim_status'] = App_Claim_Status_Production::CREATE_STATUS;  
            $data['type_id'] = App_Claim_Factory::TYPE_ID_PRODUCTION;  
            $data['claimType'] = App_Claim_Factory::CLAIM_TYPE_PRODUCTION;  
            $data['depot_id'] = App_Db_Users::obtain()->getUserOrg($data['manager_id']);  
            $data['date_begin'] = App_Formatter::getCurTime();  
            $data['is_created'] = 0;  
  
            $claim->id = App_Db::get(DB_CLAIMS)->insert($data);  
            App_Db::get(DB_CLAIMS)->update(  
                array(  
                    'full_id' => self::getFullId($claim->id),  
                    'url' => '/claim/production/' . $claim->id  
                ),  
                array('id = ?' => $claim->id)  
            );  
        } else {  
            if (count($data) > 0) {  
                if (Claim_Model_Production_Access::canEditField($claim->id, 'manager')) {  
                    $data['depot_id'] = App_Db_Users::obtain()->getUserOrg($data['manager_id']);  
                }  
                // Обновление заявки  
                App_Db::get(DB_CLAIMS)->update($data, array('id = ?' => $claim->id));  
            }  
        }  
  
        if (Claim_Model_Production_Access::canEditField($claim->id, 'client')) {  
            // Сохраняем текстовое наименование клиента в таблицу claim_production_textclientname  
            App_Db_ClaimProductionTextclientname::obtain()->save(  
                array(  
                    'claim_id' => $claim->id,  
                    'is_linked_project_client' => isset($claim->isLinkedProjectClient) ? (int) $claim->isLinkedProjectClient : 0,  
                    'client_name' => isset($claim->textClientName) ? trim($claim->textClientName) : null  
                ),  
                $claim->manager_id,  
                $claim->claimIsCopying ? $claim->oldClaimId : null // В случае  
                // копирования также передаем идентификатор копируемой заявки            );  
        }  
  
        if (Claim_Model_Production_Access::canEditField($claim->id, 'premanufacturers')) {  
            // Сохраняем производственные площадки, если есть права на это  
  
            $premanufacturerIds = $claim->getPremanufacturerId();  
            if (!empty($premanufacturerIds)) {  
                // Надо проверить, правильные ли пришли данные с формы  
                foreach ($premanufacturerIds as $key => $premanufacturerId) {  
                    if (!((int) $premanufacturerId)) {  
                        unset($premanufacturerIds[$key]);  
                    }  
                }  
                if (empty($premanufacturerIds)) {  
                    // Если по итогу не было выбрано ни одной площадки, обнуляем  
                    // поле в объекте заявки                    $claim->premanufacturerId = null;  
                }  
  
                $claim->premanufacturerId = $premanufacturerIds;  
            }  
  
            Claim_Model_Production_Premanufacturers::getInstance()->save($claim);  
        }  
  
        // рубрики товара  
        if (Claim_Model_Production_Access::canEditField($claim->id, 'item_rubric')) {  
            Claim_Model_Production_ItemRubric::getInstance()->save($claim);  
        }  
  
        // Автоматическое проставление флага "В производство" при отправке на статус 55  
        if ($claim->changeStatusAvaleble && $statusClass->getNextStatus() == App_Claim_Status_Production::IN_WORK_PENDING_STATUS) {  
            if (is_array($claim->getPremanufacturerId()) && count($claim->getPremanufacturerId()) == 1  
                && is_array($claim->getPremanufacturerMainPrice())  
                && (int) array_sum($claim->getPremanufacturerMainPrice()) == 0  
            ) {  
                App_Db::get(DB_CLAIM_PRODUCTION_PREMANUFACTURERS)->update(  
                    array(  
                        'is_main_price' => 1  
                    ),  
                    array(  
                        'claim_id = ?' => $claim->getId()  
                    )  
                );  
            }  
        }  
  
        if ($claim->changeStatusAvaleble || $claim->productionWaitingClosure) {  
            // Сохранение причины закрытия/отката заявки и примечения к ней.  
            // + отправка необходимых писем            Claim_Model_Production_Annotation::getInstance()->save($claim);  
        }  
  
        // Не забываем про товар по заявке  
        App_Claim_Mapper_Production_Product::getInstance()->updateClaimProduct($claim);  
        // И про параметры по заявке на производство  
        Claim_Model_Production_Param::getInstance()->save($claim);  
  
        // Загрузка файлов  
        if (Claim_Model_Production_Access::canEditField($claim->id, 'files')) {  
            $annotations = $claim->filesAnnotations;  
            $files = $claim->getFiles();  
  
            if (!empty($files['uploaded'])) {  
                foreach ($files['uploaded'] as $uploadedBlockKey => $uploadedBlock) {  
                    foreach ($uploadedBlock['name'] as $fileKey => $uploadedBlockFileName) {  
                        if (!$uploadedBlockFileName) {  
                            continue;  
                        }  
  
                        $fileModel = new Claim_Model_File_FileProduction();  
                        $uploadResult = $fileModel->upload(array(  
                            'claimId' => $claim->id,  
                            'files' => array(  
                                'name' => $uploadedBlock['name'][$fileKey],  
                                'type' => $uploadedBlock['type'][$fileKey],  
                                'tmp_name' => $uploadedBlock['tmp_name'][$fileKey],  
                                'error' => $uploadedBlock['error'][$fileKey],  
                                'size' => $uploadedBlock['size'][$fileKey]  
                            ),  
                            'annotation' => $annotations[$uploadedBlockKey] ? strip_tags($annotations[$uploadedBlockKey]) : null  
                        ));  
  
                        if (!$uploadResult['result'] === true) {  
                            throw new Exception($uploadResult['message']);  
                        }  
                    }  
                }  
            }  
        }  
  
        // Загрузка файлов от отдела снабжения  
        if (Claim_Model_Production_Access::canEditField($claim->id, 'filesfrompurchasingdepartment')) {  
            $annotations = $claim->filesFromPurchasingDepartmentAnnotations;  
            $files = $claim->getFilesFromPurchasingDepartment();  
  
            if (!empty($files['uploaded'])) {  
                foreach ($files['uploaded'] as $uploadedBlockKey => $uploadedBlock) {  
                    foreach ($uploadedBlock['name'] as $fileKey => $uploadedBlockFileName) {  
                        if (!$uploadedBlockFileName) {  
                            continue;  
                        }  
  
                        $fileModel = new Claim_Model_File_FileProduction_FromPurchasingDepartment();  
  
                        $uploadResult = $fileModel->upload(array(  
                            'claimId' => $claim->id,  
                            'files' => array(  
                                'name' => $uploadedBlock['name'][$fileKey],  
                                'type' => $uploadedBlock['type'][$fileKey],  
                                'tmp_name' => $uploadedBlock['tmp_name'][$fileKey],  
                                'error' => $uploadedBlock['error'][$fileKey],  
                                'size' => $uploadedBlock['size'][$fileKey]  
                            ),  
                            'annotation' => $annotations[$uploadedBlockKey] ? strip_tags($annotations[$uploadedBlockKey]) : null  
                        ));  
  
                        if (!$uploadResult['result'] === true) {  
                            throw new Exception($uploadResult['message']);  
                        }  
                    }  
                }  
            }  
        }  
  
        // Поле "Групповой заказ"  
        if (Claim_Model_Production_Access::canEditField($claim->id, 'grouporder')) {  
            if ($claim->grouppedClaimId) {  
                // Делаем проверку, но только в том случае, если "Дата подачи" не была установлена автоматически  
                $infoClaim = App_Claim_Helper_Production::getInfoClaim($claim->id);  
  
                $availableClaimsIds = array();  
                $availableClaimsList = Claim_Model_Production_GroupOrder::getInstance()->getClaimsList($claim->id, $infoClaim->managerId, date('d.m.Y', $infoClaim->date), $infoClaim->product['depot']['field1']);  
                foreach ($availableClaimsList as $availableClaim) {  
                    $availableClaimsIds[] = $availableClaim->id;  
                }  
  
                if (!in_array($claim->grouppedClaimId, $availableClaimsIds)) {  
                    if (!$resetDate) {  
                        // Генерируем ошибку, игнорируя случай, когда происходит автоматическая установка "Даты подачи"  
                        throw new RuntimeException('Неверно указан идентификатор групповой заявки. Обратитесь к администратору');  
                    } else {  
                        $claim->grouppedClaimId = null;  
                    }  
                }  
            }  
  
            $grouppedClaimId = $claim->grouppedClaimId ? $claim->grouppedClaimId : new Zend_Db_Expr('NULL');  
            App_Db::get(DB_CLAIMS)->update(array('groupped_claim_id' => $grouppedClaimId), array('id = ?' => $claim->id));  
        }  
  
        if (App_Access::get('access', 'claim>production>change_group_status') &&  
            is_array($claim->changeGroupStatus) && count($claim->changeGroupStatus)  
        ) {  
            Claim_Model_Production_GroupChangeStatus::getInstance()->changeStatus($claim, $claim->changeGroupStatus);  
        }  
  
        // сохранение настроек аренды и согласий руководителей  
        $this->_saveDepotRentDelay($claim);  
  
        // обновление флага "Крупная заявка", если rweight из claim_products больше заданного значения  
        $this->_updateRweightLimitAlert($claim);  
  
        // Отправка объявления о предоставлении БЦ по заявке.  
        if (  
            (int) $claim->claimStatus === App_Claim_Status_Production::STMANAGER_STATUS_AFTER_BASE_PRICE_IN_PROCESS_STATUS &&  
            (int) $statusClass->getNextStatus() === App_Claim_Status_Production::BASE_PRICE_PROCESSED_STATUS  
        ) {  
            Declaration_Model_DeclarationTemplate_ClaimProductionBasePriceResponse::notify($claim->id, $claim->manager_id, $claim->product);  
        }  
  
        if ($statusClass->getNextStatus() == App_Claim_Status_Production::IN_WORK_PENDING_STATUS) {  
            App_Db_Claims::obtain()->update(  
                [  
                    'date_set_in_work_pending_status' => (new DateTimeImmutable())->format(MYSQL_DATETIME_FORMAT)  
                ],  
                [  
                    'id = ?' => $claim->id  
                ]  
            );  
        }  
    }  
  
  
    /**  
     * Получение значения TASK     *     * @param Integer $claimId Идентификатор заявки  
     *     * @return string  
     */  
    public static function getFullId($claimId)  
    {  
        $task = 'TASK';  
        $zeroCols = '000000';  
        $claimIdNum = -(strlen($claimId));  
        $zeros = substr($zeroCols, 0, $claimIdNum);  
  
        return $task . $zeros . $claimId;  
    }  
  
  
    /**  
     * Выбор заявки по id     *     * @param Integer $id                       Идентификатор заявки  
     * @param Claim_Model_Production $claim     Объект заявки, в который будут записаны данные. Если NULL, будет создан новый  
     *     * @return Claim_Model_Production  
     */  
    public function find($id, ?Claim_Model_Production $claim = null)  
    {  
        $row = App_Db::get(DB_CLAIMS)->getRow(array('id = ?' => $id), Zend_Db::FETCH_OBJ);  
  
        if (!$claim) {  
            $claim = new Claim_Model_Production();  
            $claim->managerId = (integer) Zend_Auth::getInstance()->getIdentity()->id;  
            $claim->claimStatus = App_Claim_Status_Production::CREATE_STATUS;  
        }  
        if (!$row) {  
            return $claim;  
        }  
  
        // Товар по заявке  
        $row->product = App_Claim_Mapper_Production_Product::getInstance()->find($id);  
  
        // Параметры по заявке  
        $row->productionParams = Claim_Model_Production_Param::getInstance()->findParamsByClaimId($id);  
  
        $row->date = (integer) $row->date;  
  
        // Текстовое наименование клиента  
        $row->textClientName = App_Db_ClaimProductionTextclientname::obtain()->getClientNameByClaimId($id);  
  
        // Производственные площадки  
        $premanufacturersData = Claim_Model_Production_Premanufacturers::getInstance()->findPremanufacturersIdsByClaimId($id);  
        $isLinkedClaim = (  
            $id > 0 &&  
            App_Claim_LinkedClaim_Finder::getInstance()->findByLocalClaimId($id) !== false  
        );  
        if (!empty($premanufacturersData)) {  
            $row->premanufacturerId = array();  
            $row->premanufacturerName = array();  
            $row->premanufacturerPlannedBaseprice = array();  
            $row->premanufacturerPlannedBasepriceDate = array();  
            $row->premanufacturerPlannedBasepriceAnnotation = array();  
            $row->premanufacturerMainPrice = array();  
  
            foreach ($premanufacturersData as $premanufacturersItem) {  
                $row->premanufacturerId[] = $premanufacturersItem->client_id;  
                if ($isLinkedClaim) {  
                    $row->premanufacturerName[] = $premanufacturersItem->client_name;  
                }  
                $row->premanufacturerPlannedBaseprice[] = $premanufacturersItem->planned_baseprice;  
                $row->premanufacturerPlannedBasepriceDate[] = $premanufacturersItem->planned_baseprice_date;  
                $row->premanufacturerPlannedBasepriceAnnotation[] = $premanufacturersItem->planned_baseprice_annotation;  
                $row->premanufacturerMainPrice[] = $premanufacturersItem->is_main_price;  
            }  
        } else {  
            $row->premanufacturerId = null;  
            $row->premanufacturerName = null;  
            $row->premanufacturerPlannedBaseprice = null;  
            $row->premanufacturerPlannedBasepriceDate = null;  
            $row->premanufacturerPlannedBasepriceAnnotation = null;  
            $row->premanufacturerMainPrice[] = null;  
        }  
  
        $row->files = App_Db_ClaimFiles::obtain()->getFilesByClaimId($id, Claim_Model_File_File::getType('production'));  
        $row->files_from_purchasing_department = App_Db_ClaimFiles::obtain()->getFilesByClaimId($id, Claim_Model_File_File::getType('productionFromPurchasingDepartment'));  
  
        $depotRentDelay = App_Db_ClaimProductionRentDelay::obtain()->find($id)->current();  
        if ($depotRentDelay?->claim_id > 0) {  
            $row->depotRentDelayDate = $depotRentDelay->delay_date;  
            $row->depotRentDirectorApproval = $depotRentDelay->director_approval;  
            $row->depotRentDirectorComment = $depotRentDelay->director_comment;  
            $row->depotRentPurchaseDirectorApproval = $depotRentDelay->purchase_director_approval;  
            $row->depotRentPurchaseDirectorComment = $depotRentDelay->purchase_director_comment;  
            $row->depotRentManagerComment = $depotRentDelay->manager_comment;  
        }  
  
        $row->itemRubric = Claim_Model_Production_ItemRubric::getInstance()->getClaimData($id);  
  
        $claim->setOptions($row);  
  
        /** @var \App\SalesFunnel\Lead\Relation\Entity $leadRelation */  
        $leadRelation = (new \App\SalesFunnel\Lead\Relation\Repository())::find()->byClaim($id)->one(true);  
        $claim->setLeadId($leadRelation->getLeadId());  
        $claim->setLeadProjectId($leadRelation->getLeadProjectId());  
  
        $claim->setRepeatedAppeals($this->getRepeatedAppealsByClaim($id));  
        $claim->setRepeatedAppealReasons($this->getRepeatedAppealReasonsByClaim($id));  
  
        return $claim;  
    }

    /**  
     * сохранение полей, связанных с отсрочкой аренды склада     * @param Claim_Model_Production $claim  
     * @return void  
     */  
    protected function _saveDepotRentDelay(Claim_Model_Production $claim)  
    {  
        $deleteExisting = false;  
        $depotRentData = array();  
  
        $depotRentDelay = App_Db_ClaimProductionRentDelay::obtain()->find($claim->id)->current();  
  
        if (Claim_Model_Production_Access::canEditDepotRentField($claim->id, 'delay_date')  
            && !is_null($claim->depotRentDelayDate)  
        ) {  
            $depotRentData['delay_date'] = date('Y-m-d', strtotime($claim->depotRentDelayDate));  
            $depotRentData['manager_comment'] = $claim->depotRentManagerComment;  
        } else if (  
            Claim_Model_Production_Access::canEditDepotRentField($claim->id, 'delay_date')  
            && is_null($claim->depotRentDelayDate)  
        ) {  
            // если есть право на редактирование даты отсрочки и данные по отсрочке не пришли,  
            // то удаляем существующую запись            $deleteExisting = true;  
        }  
  
        if (Claim_Model_Production_Access::canEditDepotRentField($claim->id, 'director_approval')  
            && !is_null($claim->depotRentDirectorApproval)  
            && ($depotRentDelay?->delay_date > ''  
                || isset($depotRentData['delay_date'])  
            )  
        ) {  
            $depotRentData['director_approval'] = $claim->depotRentDirectorApproval;  
            $depotRentData['director_comment'] = $claim->depotRentDirectorComment;  
        }  
  
        if (Claim_Model_Production_Access::canEditDepotRentField($claim->id, 'purchase_director_approval')  
            && !is_null($claim->depotRentPurchaseDirectorApproval)  
            && ($depotRentDelay?->delay_date > ''  
                || isset($depotRentData['delay_date'])  
            )  
        ) {  
            $depotRentData['purchase_director_approval'] = $claim->depotRentPurchaseDirectorApproval;  
            $depotRentData['purchase_director_comment'] = $claim->depotRentPurchaseDirectorComment;  
        }  
  
        if (!$deleteExisting && count($depotRentData) > 0) {  
            if ($depotRentDelay?->claim_id > 0) {  
                App_Db_ClaimProductionRentDelay::obtain()->update(  
                    $depotRentData,  
                    array(  
                        'claim_id = ?' => $claim->id  
                    )  
                );  
            } else {  
                App_Db_ClaimProductionRentDelay::obtain()->insert(  
                    array_merge(  
                        array('claim_id' => $claim->id),  
                        $depotRentData  
                    )  
                );  
            }  
  
            // отправка сообщения пользователю об отклонении его предложения об аренде склада  
            if (array_key_exists('director_approval', $depotRentData) && !$depotRentData['director_approval']  
                && $depotRentDelay->director_approval != $depotRentData['director_approval']  
                || array_key_exists('purchase_director_approval', $depotRentData) && !$depotRentData['purchase_director_approval']  
                && $depotRentDelay->purchase_director_approval != $depotRentData['purchase_director_approval']  
            ) {  
                $directorName = App_Db_Users::obtain()->getUserName(App_User_Wrapper::getInstance()->getId());  
                Declaration_Model_DeclarationTemplate::get('production_depot_rent_delay_denial')->send(array(  
                    'destination_users' => array($claim->manager_id),  
                    'message' => 'По заявке ' . $claim->id . ', ' . $directorName . ' не одобрил условия аренды. В случае, если вы хотите остановить производство, обратитесь в отдел закупок'  
                ));  
            }  
        } else if ($deleteExisting) {  
            App_Db_ClaimProductionRentDelay::obtain()->delete(array(  
                'claim_id = ?' => $claim->id  
            ));  
        }  
    }  
  
  
    /**  
     * Обновляет флаг "Крупная заявка"     * @param Claim_Model_Production $claim  
     * @return void  
     */  
    protected function _updateRweightLimitAlert(Claim_Model_Production $claim)  
    {  
        $product = $claim->getProduct();  
  
        // установка названия товара  
        $field1 = $product['depot']['field1'];  
  
        // установка сорт товара  
        $field2 = $product['depot']['field2'];  
  
        // установка расчетного веса товара  
        $rWeight = $product['rweight'];  
  
        // установка количества товара  
        $amount = $product['amount'];  
  
        // проверка условий для товаров  
        switch (true) {  
            case (preg_match('/Пакет/', $field1) && $rWeight > 2000):  
            case (preg_match('/Майка/', $field1) && $rWeight > 2000):  
            case (preg_match('/Пленка/', $field1) && $rWeight > 3000):  
            case (preg_match('/Рукав/', $field1) && $rWeight > 3000):  
            case (preg_match('/Полурукав/', $field1) && $rWeight > 3000):  
            case (preg_match('/Полотно/', $field1) && $rWeight > 3000):  
            case (preg_match('/Стрейч/', $field1) && preg_match('/вторичка/', $field2) && $rWeight > 3000):  
            case (preg_match('/Скотч/', $field1) && $amount > 3000):  
                $limitAlert = 1;  
                break;  
            default:  
                $limitAlert = 0;  
        }  
  
        // обновление флага rweight_limit_alert в базе  
        App_Db_Claims::obtain()->obtain()->update(  
            array('rweight_limit_alert' => $limitAlert),  
            array('id = ?' => $claim->id)  
        );  
    }  
  
    /**  
     * Получить список сущностей повторных обращений по id заявки     * @param int $claimId  
     * @return RepeatedAppealModel[]  
     */  
    private function getRepeatedAppealsByClaim(int $claimId): array  
    {  
        $condition = new RepeatedAppealConditions();  
        $condition->getClaimId()->equal($claimId);  
        $condition->orderById();  
  
        return $this->repeatedAppealsRepository->findBy($condition);  
    }  
  
    /**  
     * Получить причины повторных обращений по id заявки     * @param int $claimId  
     * @return RepeatedAppealReasonModel[]  
     */  
    private function getRepeatedAppealReasonsByClaim(int $claimId): array  
    {  
        $condition = new RepeatedAppealReasonConditions();  
        $condition->getClaimId()->equal($claimId);  
        $condition->orderById();  
  
        return $this->repeatedAppealReasonsRepository->findBy($condition);  
    }
