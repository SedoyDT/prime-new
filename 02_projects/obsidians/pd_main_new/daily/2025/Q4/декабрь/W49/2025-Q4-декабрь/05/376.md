DROP TABLE IF EXISTS `sales_funnel_lead_activity_event`;  
CREATE TABLE `sales_funnel_lead_activity_event` (  
    `id` INT AUTO_INCREMENT PRIMARY KEY,  
    `event_name` VARCHAR(150) NOT NULL COMMENT 'Название события',  
    `event_description` VARCHAR(1000) COMMENT 'Описание события',  
    `is_active` TINYINT UNSIGNED NOT NULL DEFAULT 1 COMMENT 'считается ли это событие активностью',  
    INDEX (event_name)  
) COMMENT='Воронка продаж: Указания какие события являются активностями';  
  
DROP TABLE IF EXISTS `sales_funnel_lead_activity_field`;  
CREATE TABLE `sales_funnel_lead_activity_field` (  
       `id` INT AUTO_INCREMENT PRIMARY KEY,  
       `field` VARCHAR(50) NOT NULL COMMENT 'field, поле в лиде',  
       `field_title` VARCHAR(50) NOT NULL COMMENT 'Понятное человеку название поля лида',  
       `is_active` TINYINT UNSIGNED NOT NULL DEFAULT 1 COMMENT 'считается ли изменение этого поля активностью',  
       UNIQUE KEY `field` (`field`)  
) COMMENT = 'Воронка продаж: настройка учёта изменений полей лида как активности';
