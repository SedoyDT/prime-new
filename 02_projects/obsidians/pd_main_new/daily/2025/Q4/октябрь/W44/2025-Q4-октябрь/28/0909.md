var contract = new (class extends Libdoc_Contracts_Editor_Contract_Abstract2
{
    init(data, specificationData, contractId)
    {
        var _self = this;

        this._initData();
        this.specificationProductsData = specificationData;
        this.contractId = contractId;

        // Модификация настроек редактора
        data = $.extend({
            save: {
                onClick: function (element)
                {
                    // Безопасный вызов collectData
                    try {
                        if (window.structure && window.structure.save) {
                            var collectedData = window.structure.save.collectData();
                            
                            // Исправляем contractDate если он undefined
                            if (!collectedData.contractDate) {
                                var datePicker = $('#contractDatePickerButton');
                                if (datePicker.length) {
                                    collectedData.contractDate = datePicker.val();
                                }
                            }
                            
                            window.structure.save.saveRequest(collectedData);
                        }
                    } catch (error) {
                        console.error('Error during save:', error);
                        alert('Ошибка при сохранении: ' + error.message);
                    }
                }
            }
        }, data);

        // Остальной код без изменений...
        if (window.structure && typeof window.structure.init === 'function') {
            window.structure.init(data);
        }

        this.bindEvents();
        this.setSpecifications();
        this.updateFields();

        this.checkboxTextObjects = {};
        this.radioTextObjects = {};

        if (super.init) {
            super.init(data, specificationData, contractId);
        }
        
        this.updateExceptionVATFields();
        
        window.structure.onAfterUpdateFields(() => {
            this.updateExceptionVATFields();
        });
    }

    // ... остальные методы
});
