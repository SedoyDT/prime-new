actor "Пользователь" as User

rectangle "Уровень представления" {
    User -> "Генерация запроса" : Направляет запрос\nна уровень управления
    "Генерация запроса" -> "Оценка результатов" : Интерпретирует запрос\nи выбирает правильное\nпредставление

    "Генерация запроса" --> "Интерпретация запроса" : Передает запрос\nна обработку
    "Интерпретация запроса" -> "Логика приложения" : Запрашивает решение\nпоставленной задачи
    "Логика приложения" --> "Интерпретация запроса" : Возвращает результаты
    "Интерпретация запроса" --> "Оценка результатов" : Передает результаты\nдля отображения
}

rectangle "Уровень данных" {
    "Логика приложения" -> "Управление данными" : Управляет получением\nи сохранением данных
    "Управление данными" --> "Логика приложения" : Предоставляет данные
}
