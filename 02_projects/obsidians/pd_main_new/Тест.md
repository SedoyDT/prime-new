    public function testLeadCommandToSendApiRequest()
    {
        $behaviour = $this->container->make(\App\SalesFunnel\LeadProp\PropValue\Behaviour\GetBehaviour::class);
        $leadPhoneFactory = $this->container->make(\App\SalesFunnel\Lead\Phone\Services\Factory::class);

        $handler = new SendApiRequestHandler(
            $behaviour,
            $leadPhoneFactory,
        );

        $stageId = null;
        if (APPLICATION_ENV == 'pzo') {
            $stageId = self::PZO_STAGE_ID;
        } else if (APPLICATION_ENV == 'amdcoru') {
            $stageId = self::AMDCORU_STAGE_ID;
        }

        $leadCollectionByStage = $this->leadFactoryMock->findByStageId($stageId);
        $triggers = $this->stageTriggerFactory->collectionFromStage($stageId);

        foreach ($triggers as $trigger) {
            $command = $trigger->getCommand();
            $command->setTrigger($trigger);
        }

        $command->setLeadCollection($leadCollectionByStage);

        $result = $handler->handle($command);

        foreach ($result as $statusCode) {
            $this->assertEquals(self::EXPECTED_SUCCESS_STATUS_CODE, $statusCode);
        }

    }
