```sql
# Не пишите запросы на удаление таблиц в обычных задачах.  
# В данном руководстве они используются для быстрого сброса состояния базы.  
DROP TABLE IF EXISTS counteragent_phone;  
DROP TABLE IF EXISTS counteragent;  
DROP TABLE IF EXISTS counteragent_status;  
DROP TABLE IF EXISTS counteragent_type;  
  
create table counteragent_status  
(  
    id    int unsigned auto_increment comment 'Id статуса' primary key,  
    title char(100) not null comment 'Название статуса',  
    alias char(100) not null comment 'Название константы',  
    constraint counteragent_status_pk_1 unique (title),  
    constraint counteragent_status_pk_2 unique (alias)  
)  
    comment 'Статусы контрагента';  
  
INSERT INTO counteragent_status (id, title, alias)  
VALUES  
    (1, 'Действующий', 'active'),  
    (2, 'Ликвидирован', 'liquidated'),  
    (3, 'В процессе ликвидации', 'liquidating')  
;  
  
create table counteragent_type  
(  
    id    int unsigned auto_increment comment 'Id типа' primary key,  
    title char(100) not null comment 'Название типа',  
    alias char(100) not null comment 'Название константы',  
    constraint customer_type_pk_1 unique (title),  
    constraint customer_type_pk_2 unique (alias)  
)  
    comment 'Типы контрагентов';  
  
INSERT INTO counteragent_type (id, title, alias)  
VALUES  
    (1, 'Покупатель', 'customer'),  
    (2, 'Поставщик', 'supplier')  
;  
  
create table counteragent  
(  
    id                int unsigned auto_increment comment 'Id контрагента' primary key,  
    title             char(100)    not null comment 'Название контрагента',  
    inn               char(100)    not null comment 'ИНН контрагента',  
    established_at    date         null comment 'Дата регистрации в ФНС',  
    account_number    int          null comment 'Номер счёта',  
    has_edm           tinyint(1)   not null comment 'Работает с ЭДО?',  
    type_id           int unsigned not null comment 'Тип из counteragent_type',  
    status_id         int unsigned not null comment 'Статус из counteragent_status',  
    comment           text         null comment 'Комментарий',  
    constraint counteragent_fk_type foreign key (type_id) references counteragent_type (id),  
    constraint counteragent_fk_status foreign key (status_id) references counteragent_status (id)  
)  
    comment 'Контрагент';  
  
create table counteragent_phone  
(  
    id              int unsigned auto_increment comment 'Id номера' primary key,  
    counteragent_id int unsigned    not null comment 'Id контрагента',  
    number          bigint unsigned not null comment 'Номер телефона',  
    comment         text            null comment 'Комментарий',  
    constraint counteragent_phone_fk_counteragent foreign key (counteragent_id) references counteragent (id) on delete cascade  
)  
    comment 'Номера контрагента';
```
