## Месяцы
```dataviewjs
const file = dv.current().file;
const created = file.ctime;

// текущий квартал от даты создания файла
const year = created.year;
const quarter = Math.ceil(created.month / 3);

// путь квартала
const base = `${year}/${year}-Q${quarter}`;

// месяцы текущего квартала
const months = [];

for (let month = 1; month <= 12; month++) {
    const q = Math.ceil(month / 3);

    if (q === quarter) {
        const mm = String(month).padStart(2, "0");

        // YYYY/YYYY-QX/YYYY-MM/YYYY-MM
        const filePath = `${base}/${year}-${mm}/${year}-${mm}`;

        months.push([
            dv.fileLink(filePath)
        ]);
    }
}

dv.table(["Month"], months);
```
